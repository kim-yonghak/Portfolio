  function filterMapTree(){
    const q = (document.getElementById('mapFilter')?.value || "").trim().toLowerCase();
    if(!q){ renderMapTree(MAP_TREE_DATA); return; }
    const filtered = [];
    for(const m of MAP_TREE_DATA){
      const mapName = String(m.map || "");
      const newHunts = [];
      for(const h of (m.hunts || [])){
        const huntName = String(h.hunt || "");
        const monsters = (h.monsters || []).filter(mon => `${mapName} ${huntName} ${mon.id} ${mon.name}`.toLowerCase().includes(q));
        const huntMatch = `${mapName} ${huntName}`.toLowerCase().includes(q);
        if(huntMatch || monsters.length){
          const finalMonsters = huntMatch ? (h.monsters || []) : monsters;
          newHunts.push({...h, monsters: finalMonsters, monster_count: finalMonsters.length});
        }
      }
      if(newHunts.length){
        filtered.push({...m, hunts:newHunts, hunt_count:newHunts.length, monster_count:newHunts.reduce((a,h)=>a+(h.monster_count||0),0)});
      }
    }
    renderMapTree(filtered, true);
  }

  function onKey(ev){
    const box = document.getElementById('aclist');
    const opened = (box.style.display === "block");

    if(ev.key === "Enter"){
      if(opened && acIndex >= 0 && acIndex < acItems.length){
        ev.preventDefault();
        document.getElementById('q').value = String(acItems[acIndex].name);
        box.style.display = "none";
        doSearch(acItems[acIndex].id);
        acIndex = -1;
        return;
      }
      if(opened){
        box.style.display = "none";
        acIndex = -1;
      }
      ev.preventDefault();
      doSearch();
      return;
    }

    if(!opened) return;

    if(ev.key === "ArrowDown"){
      ev.preventDefault();
      acIndex = Math.min(acItems.length-1, acIndex+1);
    } else if(ev.key === "ArrowUp"){
      ev.preventDefault();
      acIndex = Math.max(-1, acIndex-1);
    } else if(ev.key === "Escape"){
      box.style.display = "none";
      acIndex = -1;
    }

    [...box.children].forEach((c, i) => {
      c.style.background = (i === acIndex) ? "#eef2ff" : "white";
    });
  }

  function updateParentState(parentChk, childChks){
    const total = childChks.length;
    let checked = 0;
    let partial = false;

    childChks.forEach(ch => {
      if(ch.dataset.state === "partial") partial = true;
      else if(ch.checked) checked += 1;
    });

    if(checked === total){
      parentChk.checked = true;
      parentChk.indeterminate = false;
      parentChk.dataset.state = "checked";
    } else if(checked === 0 && !partial){
      parentChk.checked = false;
      parentChk.indeterminate = false;
      parentChk.dataset.state = "unchecked";
    } else {
      parentChk.checked = false;
      parentChk.indeterminate = true;
      parentChk.dataset.state = "partial";
    }
  }

  function fmtInt(v){
    if(v == null || v === "") return "—";
    const n = Number(v);
    if(!Number.isFinite(n)) return "—";
    return n.toLocaleString('en-US');
  }

  function fmtProb(p){
    if(p == null || p === "") return "—";
    const n = Number(p);
    if(!Number.isFinite(n)) return "—";
    const pct = n * 100.0;
    const txt = (Math.round(pct * 10000) / 10000).toString();
    return `${txt}%`;
  }

  function clsDelta(x){
    if(x == null || x === "") return "";
    const n = Number(x);
    if(!Number.isFinite(n)) return "";
    return (n >= 0) ? "currencyDeltaPos" : "currencyDeltaNeg";
  }

  function fmtDeltaInt(bMinusA){
    if(bMinusA == null || bMinusA === "") return "—";
    const n = Number(bMinusA);
    if(!Number.isFinite(n)) return "—";
    const sign = (n >= 0) ? "+" : "";
    return sign + fmtInt(n);
  }

  function fmtDeltaProb(bMinusA){
    if(bMinusA == null || bMinusA === "") return "—";
    const n = Number(bMinusA);
    if(!Number.isFinite(n)) return "—";
    const pct = n * 100.0;
    const sign = (pct >= 0) ? "+" : "";
    const txt = (Math.round(pct * 10000) / 10000).toString();
    return `${sign}${txt}%`;
  }

  function compareNpcStatValues(av, bv, loadedA, loadedB){
    const presentA = av != null && av !== "";
    const presentB = bv != null && bv !== "";
    if(!loadedA || !loadedB) return {label:"비교 불가", changed:false, delta:null};
    if(!presentA && !presentB) return {label:"데이터 없음", changed:false, delta:null};
    if(!presentA) return {label:"추가", changed:true, delta:null};
    if(!presentB) return {label:"삭제", changed:true, delta:null};
    const delta = Number(bv) - Number(av);
    if(!Number.isFinite(delta)) return {label:"비교 불가", changed:false, delta:null};
    return {label:delta === 0 ? "동일" : fmtDeltaInt(delta), changed:delta !== 0, delta};
  }

  function normalizeSearchResults(payload){
    const out = [];
    const src = payload.results || payload.entries || [];

    src.forEach(node => {
      if(node.type === "set"){
        out.push({
          type: "set",
          text: node.text || "[SET]",
          children: (node.children || []).map(ch => normalizeItemNode(ch))
        });
      }else if(node.type === "item"){
        out.push({ type: "item", ...normalizeItemNode(node.row || node) });
      }
    });

    if(out.length === 0 && Array.isArray(payload.items)){
      payload.items.forEach(item => out.push({ type:"item", ...normalizeItemNode(item) }));
    }

    return out;
  }

  function normalizeItemNode(raw){
    raw = raw || {};
    const id = raw.id ?? raw.item_info_id ?? raw["Item_Info IDX"] ?? raw.Item_Info_IDX ?? "";
    const name = raw.name ?? raw.item_name ?? raw.item_string ?? raw.Item_String ?? raw["Item_String"] ?? `Unknown Item(${id})`;
    const probTxt = raw.prob_txt ?? raw.prob_text ?? raw.DropProbText ?? raw.drop_prob_text ?? raw["DropProbText"] ?? raw["prob_text"];
    const probVal = raw.prob ?? raw.drop_prob ?? raw.DropProb ?? raw["DropProb"];
    return {
      id,
      name,
      prob: probVal,
      prob_txt: probTxt || fmtDisplayProb(probVal),
      min_amount: raw.min_amount ?? raw.minAmount ?? raw.MinAmount,
      max_amount: raw.max_amount ?? raw.maxAmount ?? raw.MaxAmount,
      raw
    };
  }

  function fmtDisplayProb(p){
    if(p == null || p === "") return "—";
    const n = Number(p);
    if(!Number.isFinite(n)) return "—";
    const txt = (Math.round(n * 1000000) / 1000000).toString();
    return `${txt}%`;
  }

  function calcSummaryFromSearch(payload){
    if(payload.summary) return payload.summary;
    const rows = normalizeSearchResults(payload);
    let sets = 0;
    let items = 0;
    rows.forEach(x => x.type === "set" ? sets++ : items++);
    return { total: rows.length, sets, items };
  }

  function normalizeRandomBoxItem(raw){
    raw = raw || {};
    const id = raw.item_info_id ?? raw.id ?? "";
    let name = raw.item_name ?? raw.name ?? `Unknown Item(${id})`;
    if(String(name).startsWith("Unknown Item(") && raw.nested_box_name){
      name = raw.nested_box_name;
    }
    const nestedItems = (raw.nested_items || []).map(normalizeRandomBoxItem);
    return {
      id,
      name,
      min_amount: raw.min_amount ?? raw.min ?? raw.MinValue,
      max_amount: raw.max_amount ?? raw.max ?? raw.MaxValue,
      prob_txt: raw.prob_text ?? raw.prob_txt ?? fmtDisplayProb(raw.prob),
      group_id: raw.group_id ?? "",
      matched_name_id: raw.matched_name_id ?? "",
      is_nested_box: !!raw.is_nested_box,
      nested_box_id: raw.nested_box_id ?? "",
      nested_box_name: raw.nested_box_name ?? "",
      nested_items: nestedItems,
      raw
    };
  }

  function renderRandomBox(payload){

    function appendRandomBoxRow(item, depth=0){

        left.appendChild(spacer);

        const originalAppendChild = tree.appendChild.bind(tree);
        tree.appendChild = (node) => childrenBox.appendChild(node);

        tree.appendChild = originalAppendChild;

        openBtn.onclick = (ev) => {
          ev.stopPropagation();
          opened = !opened;
          childrenBox.style.display = opened ? "block" : "none";
          openBtn.style.transform = opened ? "rotate(90deg)" : "rotate(0deg)";

      empty.style.color = "var(--muted)";

    const box = document.getElementById('aclistRandomBox');

  async function onRandomBoxType(){
    const q = document.getElementById('qRandomBox').value.trim();
    randomBoxAcIndex = -1;

    if(q.length < 1){
      showRandomBoxAC([]);
      return;
    }

    const r = await fetch(`/api/random_box/autocomplete?q=${encodeURIComponent(q)}&limit=30&src=B&region=${encodeURIComponent(REGION)}`, { cache:'no-store' });
    const j = await r.json();
    randomBoxAcItems = (j.items || []);
    showRandomBoxAC(randomBoxAcItems);
  }

  function onRandomBoxKey(ev){
    const box = document.getElementById('aclistRandomBox');
    const opened = box && box.style.display === "block";

    if(ev.key === "Enter"){
      if(opened && randomBoxAcIndex >= 0 && randomBoxAcIndex < randomBoxAcItems.length){
        ev.preventDefault();
        const picked = randomBoxAcItems[randomBoxAcIndex];
        document.getElementById('qRandomBox').value = String(picked.name || picked.id);
        box.style.display = "none";
        doRandomBoxSearch(picked.id);
        randomBoxAcIndex = -1;
        return;
      }

      if(opened){
        box.style.display = "none";
        randomBoxAcIndex = -1;
      }

      ev.preventDefault();
      doRandomBoxSearch();
      return;
    }

    if(!opened) return;

    if(ev.key === "ArrowDown"){
      ev.preventDefault();
      randomBoxAcIndex = Math.min(randomBoxAcItems.length - 1, randomBoxAcIndex + 1);
    }else if(ev.key === "ArrowUp"){
      ev.preventDefault();
      randomBoxAcIndex = Math.max(-1, randomBoxAcIndex - 1);
    }else if(ev.key === "Escape"){
      box.style.display = "none";
      randomBoxAcIndex = -1;
    }

    [...box.children].forEach((c, i) => {
      c.style.background = (i === randomBoxAcIndex) ? "#eef2ff" : "white";
    });
  }

  let CHANGE_FILTER = "all";

  function setChangeFilter(filter){
    CHANGE_FILTER = filter || "all";
    curState().changeFilter = CHANGE_FILTER;

    document.querySelectorAll('#changeFilters .changeFilter').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.filter === CHANGE_FILTER);
    });

    if(curState().changesPayload){
      renderAllChanges(curState().changesPayload);
    }
  }

  function changeMatchesFilter(row, filter){
    if(filter === "all") return true;

    const type = String(row.change_type || "");
    if(filter === "probability") return type.includes("probability");
    if(filter === "added") return type.includes("added");
    if(filter === "removed") return type.includes("removed");
    if(filter === "amount") return type.includes("amount");
    if(filter === "currency") return type === "currency_changed";
    if(filter === "exp") return type === "exp_changed";
    if(filter === "antimagic") return type === "antimagic_changed";
    if(filter === "randombox") return row.category === "randombox";
    return true;
  }

  function formatChangeProbability(v){
    if(v == null || v === "") return "—";
    const n = Number(v);
    if(!Number.isFinite(n)) return "—";
    return `${n.toFixed(12).replace(/0+$/, '').replace(/\.$/, '')}%`;
  }

  function makeChangeLine(key, value){
    const line = document.createElement('div');
    line.className = "changeLine";

    const k = document.createElement('span');
    k.className = "changeKey";
    k.textContent = key;

    const v = document.createElement('span');
    v.className = "changeValue";
    v.textContent = value;

    line.appendChild(k);
    line.appendChild(v);
    return line;
  }

  function renderAllChanges(payload){
    const box = document.getElementById('changesList');
    const stat = document.getElementById('changesStat');
    if(!box || !stat) return;

    CHANGE_FILTER = curState().changeFilter || CHANGE_FILTER || "all";
    document.querySelectorAll('#changeFilters .changeFilter').forEach(btn => {
      btn.classList.toggle('active', btn.dataset.filter === CHANGE_FILTER);
    });

    const all = Array.isArray(payload?.changes) ? payload.changes : [];
    const rows = all.filter(row => changeMatchesFilter(row, CHANGE_FILTER));
    const summary = payload?.summary || {};

    stat.textContent =
      `변경 대상 ${summary.changed_targets || 0}개 · 전체 변경 ${summary.total_changes || 0}건 · 현재 필터 ${rows.length}건` +
      (payload?.cached ? " · 캐시 사용" : " · 새로 계산");

    box.innerHTML = "";

    if(!rows.length){
      box.innerHTML = `<div class="small">선택한 조건에 해당하는 변경사항이 없습니다.</div>`;
      return;
    }

    rows.forEach(row => {
      const card = document.createElement('div');
      card.className = "changeCard";

      const head = document.createElement('div');
      head.className = "changeCardHead";

      const title = document.createElement('div');
      title.className = "changeCardTitle";
      const prefix = row.category === "randombox" ? "상자" : "몬스터";
      title.textContent = `[${row.target_id}] ${row.target_name} · ${prefix}`;

      const type = document.createElement('span');
      type.className = "changeType";
      type.textContent = row.label || row.change_type || "변경";

      head.appendChild(title);
      head.appendChild(type);

      const body = document.createElement('div');
      body.className = "changeBody";

      if(row.map_hunt?.display){
        body.appendChild(makeChangeLine("지역", row.map_hunt.display));
      }

      if(row.item_info_id != null){
        body.appendChild(makeChangeLine("아이템", `[${row.item_info_id}] ${row.item_name || ""}`));
      }

      if("probA" in row || "probB" in row){
        body.appendChild(makeChangeLine(
          "확률",
          `A ${formatChangeProbability(row.probA)} → B ${formatChangeProbability(row.probB)}` +
          (row.delta != null ? ` (Δ ${formatChangeProbability(row.delta)})` : "")
        ));
      }

      if("minA" in row || "maxA" in row || "minB" in row || "maxB" in row){
        body.appendChild(makeChangeLine(
          "수량",
          `A ${fmtInt(row.minA)} ~ ${fmtInt(row.maxA)} → B ${fmtInt(row.minB)} ~ ${fmtInt(row.maxB)}`
        ));
      }

      if(row.currencyA !== undefined || row.currencyB !== undefined){
        const a = row.currencyA || {};
        const b = row.currencyB || {};
        body.appendChild(makeChangeLine(
          "재화",
          `A ${fmtInt(a.min_amount)} ~ ${fmtInt(a.max_amount)} / ${a.prob_text || "—"} → ` +
          `B ${fmtInt(b.min_amount)} ~ ${fmtInt(b.max_amount)} / ${b.prob_text || "—"}`
        ));
      }

      if(row.expA !== undefined || row.expB !== undefined){
        const a = row.expA || {};
        const b = row.expB || {};
        body.appendChild(makeChangeLine(
          "경험치",
          `A Lv.${fmtInt(a.level)} ${fmtInt(a.exp_min)} ~ ${fmtInt(a.exp_max)} → ` +
          `B Lv.${fmtInt(b.level)} ${fmtInt(b.exp_min)} ~ ${fmtInt(b.exp_max)}`
        ));
      }

      if(row.antimagicA !== undefined || row.antimagicB !== undefined){
        body.appendChild(makeChangeLine(
          "항마력",
          `A ${fmtInt(row.antimagicA)} → B ${fmtInt(row.antimagicB)}`
        ));
      }

      card.appendChild(head);
      card.appendChild(body);
      box.appendChild(card);
    });
  }

  async function loadAllChanges(force=false){
    const stat = document.getElementById('changesStat');
    const box = document.getElementById('changesList');

    stat.textContent = force ? "전체 변경사항 강제 재계산 중…" : "전체 변경사항 계산 중…";
    box.innerHTML = `<div class="small">A/B 전체 데이터를 비교하고 있습니다.</div>`;

    try{
      const r = await fetch(`/api/changes?region=${encodeURIComponent(REGION)}&force=${force ? "true" : "false"}`, {
        cache:"no-store"
      });
      const j = await r.json();

      if(!j.ok){
        throw new Error(j.error || "변경사항 계산 실패");
      }

      curState().changesPayload = j;
      curState().tab = "changes";
      renderAllChanges(j);
    }catch(e){
      stat.textContent = `오류: ${e?.message || e}`;
      box.innerHTML = `<div class="small">변경사항을 불러오지 못했습니다.</div>`;
    }
  }
